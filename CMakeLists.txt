cmake_minimum_required(VERSION 3.16)
project(FLASKRRTConnect VERSION 0.3.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(FLASK_AUDIT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/rrtpff" CACHE PATH "Header snapshot used by theory audit tests")
set(FLASK_EIGEN_SOURCE "" CACHE PATH "Optional offline Eigen source tree")
set(FLASK_YAML_SOURCE "" CACHE PATH "Optional offline yaml-cpp source tree")
option(FLASK_ENABLE_AVX2 "Build runtime-dispatched AVX2 collision kernel" ON)
if("${CMAKE_CXX_FLAGS} ${CMAKE_CXX_FLAGS_RELEASE}" MATCHES "-ffast-math|-Ofast|/fp:fast")
  message(FATAL_ERROR "FLASK validity checks require IEEE finite/NaN checks")
endif()
if(NOT EXISTS "${FLASK_AUDIT_SOURCE}/pff/pff_integrator.h")
  message(FATAL_ERROR "FLASK_AUDIT_SOURCE must contain the audited integrator and DF headers")
endif()
if(FLASK_EIGEN_SOURCE)
  add_library(theory_eigen INTERFACE)
  target_include_directories(theory_eigen SYSTEM INTERFACE "${FLASK_EIGEN_SOURCE}")
  add_library(Eigen3::Eigen ALIAS theory_eigen)
else()
  find_package(Eigen3 3.3 REQUIRED NO_MODULE)
endif()
if(FLASK_YAML_SOURCE)
  if(CMAKE_VERSION VERSION_GREATER_EQUAL "4.0" AND NOT CMAKE_POLICY_VERSION_MINIMUM)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
  endif()
  set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
  add_subdirectory("${FLASK_YAML_SOURCE}" "${CMAKE_BINARY_DIR}/yaml-cpp" EXCLUDE_FROM_ALL)
else()
  find_package(yaml-cpp REQUIRED)
endif()
add_library(flask_runtime STATIC src/geometry.cpp src/io.cpp)
target_include_directories(flask_runtime PUBLIC include vendor/rrtpff)
target_link_libraries(flask_runtime PUBLIC Eigen3::Eigen yaml-cpp)
if(MSVC)
  target_compile_options(flask_runtime PUBLIC /fp:strict)
else()
  target_compile_options(flask_runtime PUBLIC -ffp-contract=off)
endif()
if(FLASK_ENABLE_AVX2 AND CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|i.86")
  target_sources(flask_runtime PRIVATE src/geometry_avx2.cpp)
  target_compile_definitions(flask_runtime PRIVATE FLASK_HAS_AVX2=1)
  if(MSVC)
    set_source_files_properties(src/geometry_avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
  else()
    set_source_files_properties(src/geometry_avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-ffp-contract=off")
  endif()
endif()
add_executable(flask_plan src/main.cpp src/run_integrator.cpp src/run_unicycle.cpp src/run_quad2.cpp src/run_quad3.cpp)
target_link_libraries(flask_plan PRIVATE flask_runtime)
include(CTest)
foreach(name controllability gramian_scaling theory_assumptions)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_include_directories(test_${name} PRIVATE "${FLASK_AUDIT_SOURCE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests")
  target_link_libraries(test_${name} PRIVATE Eigen3::Eigen)
  if(MSVC)
    target_compile_options(test_${name} PRIVATE /W4 /fp:strict)
  else()
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
  endif()
  add_test(NAME ${name} COMMAND test_${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
# Runtime dimensions supplement the default sweep; no compile-time order cap.
add_test(NAME controllability_custom COMMAND test_controllability 5 7)
add_test(NAME gramian_scaling_custom COMMAND test_gramian_scaling 5 7)
set_tests_properties(controllability_custom gramian_scaling_custom PROPERTIES TIMEOUT 30)
foreach(name bvp planner validation models io)
  add_executable(test_${name} tests/test_${name}.cpp)
  target_include_directories(test_${name} PRIVATE tests)
  target_link_libraries(test_${name} PRIVATE flask_runtime)
  add_test(NAME ${name} COMMAND test_${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endforeach()
add_test(NAME cli_integrator COMMAND flask_plan "${CMAKE_CURRENT_SOURCE_DIR}/examples/integrator_obstacle.yaml" "${CMAKE_BINARY_DIR}/cli_integrator.yaml")
add_test(NAME cli_unicycle COMMAND flask_plan "${CMAKE_CURRENT_SOURCE_DIR}/examples/unicycle_straight.yaml" "${CMAKE_BINARY_DIR}/cli_unicycle.yaml")
add_test(NAME cli_quad3d COMMAND flask_plan "${CMAKE_CURRENT_SOURCE_DIR}/examples/quad3d_full.yaml" "${CMAKE_BINARY_DIR}/cli_quad3d.yaml")
set_tests_properties(cli_integrator cli_unicycle cli_quad3d PROPERTIES TIMEOUT 30)
foreach(scene uni_bugtrap uni_wall quad2d_bugtrap quad2d_hole quad3d_block quad3d_window)
  add_test(NAME dynobench_${scene} COMMAND flask_plan
    "${CMAKE_CURRENT_SOURCE_DIR}/examples/dynobench_${scene}.yaml"
    "${CMAKE_BINARY_DIR}/dynobench_${scene}.yaml" 41)
  set_tests_properties(dynobench_${scene} PROPERTIES TIMEOUT 15 LABELS "dynobench_r2")
endforeach()
